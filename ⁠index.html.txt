    function formatTimer(sec) {
      const m = String(Math.floor(sec / 60)).padStart(2, "0");
      const s = String(sec % 60).padStart(2, "0");
      return `${m}:${s}`;
    }

    function startExamTimer() {

          clearInterval(state.timerId);

            "Authorization": `Bearer ${SUPABASE_CONFIG.anonKey}`
          },
          body: JSON.stringify({ p_code: code })
        });

        if (!res.ok) {
          msgEl.style.color = "var(--err)";
          msgEl.textContent = tr.vipInvalid;
          return;
        }

        const isValid = await res.json();
        if (isValid === true) {
          state.isVip = true;
          saveState();
          msgEl.style.color = "var(--ok)";
          msgEl.textContent = tr.vipSuccess;
          renderAll();
        } else {
          msgEl.style.color = "var(--err)";
          msgEl.textContent = tr.vipInvalid;
        }
      } catch (e) {
        msgEl.style.color = "var(--err)";
        msgEl.textContent = tr.vipInvalid;
      }
    }

    document.getElementById("langSelect").addEventListener("change", (e) => {
      state.lang = e.target.value;
      renderAll();
    });

    document.getElementById("modePracticeBtn").addEventListener("click", () => {
      state.mode = "practice";
      stopExamTimer();
      renderAll();
    });

    document.getElementById("modeExamBtn").addEventListener("click", () => {
      state.mode = "exam";
      startExamTimer();
      renderAll();
    });

    document.getElementById("vipToggleBtn").addEventListener("click", () => {
      const panel = document.getElementById("vipActivationPanel");
      panel.style.display = panel.style.display === "none" ? "block" : "none";
    });

    document.getElementById("verifyVipBtn").addEventListener("click", verifyVipViaSupabase);
    document.getElementById("btnTrue").addEventListener("click", () => answerQuestion(true));
    document.getElementById("btnFalse").addEventListener("click", () => answerQuestion(false));
    document.getElementById("finishExamBtn").addEventListener("click", finishExam);

    document.getElementById("prevQBtn").addEventListener("click", () => {
      const unlocked = CHAPTERS.filter((c) => !c.vip || state.isVip);
      const idx = unlocked.findIndex((c) => c.id === state.chapterId);
      if (idx > 0) {
        state.chapterId = unlocked[idx - 1].id;
        renderAll();
      }
    });

    document.getElementById("nextQBtn").addEventListener("click", () => {
      const unlocked = CHAPTERS.filter((c) => !c.vip || state.isVip);
      const idx = unlocked.findIndex((c) => c.id === state.chapterId);
      if (idx < unlocked.length - 1) {
        state.chapterId = unlocked[idx + 1].id;
        renderAll();
      }
    });

    renderAll();
